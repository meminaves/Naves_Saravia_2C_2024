344void controlarServosYRGB()
345{
346 switch (ESTADO_ACTUAL_PUERTAS)
347 {
348 // Puertas cerradas
349
350 case AMBAS_CERRADAS:
351 /* LED 1 (verde) encendido
352 SERVO_1 OPEN -> Puerta 1 cerrada
353 SERVO_2 OPEN -> Puerta 2 cerrada */
354
355 NeoPixelAllColor(NEOPIXEL_COLOR_GREEN);
356
357 abrirServo(SERVO_1);
358 abrirServo(SERVO_2);
359
360 printf("Ambas puertas cerradas\n");
361
362 break;
363
364 // Una puerta abierta
365 case UNA_CERRADA:
366 /*
367 Luz amarilla
368 Puerta 1 abierta -> SERVO_2 cerrado
369 o bien
370 Puerta 2 abierta -> SERVO_1 cerrado
371 */
372 NeoPixelAllColor(NEOPIXEL_COLOR_YELLOW);
373 printf("Una abierta\n");
374 if (FC1 == FC_abierto) // Puerta 1 abierta
375 {
376 cerrarServo(SERVO_2);
377
378 // A) SERVO1 OPEN y SERVO2 CLOSED
379 }
380 if (FC2 == FC_abierto) // Puerta 2 abierta
381 {
382 // B) SERVO1 CLOSED y SERVO2 OPEN
383
384 cerrarServo(SERVO_1);
385 }
386 break;
387 // Dos puertas abiertas
388 case AMBAS_ABIERTAS:
389 /*
390 Error en el sistema, posible falla en el dif de presiones
391 */
392 printf("Ambas abiertas\n");
393 NeoPixelAllColor(NEOPIXEL_COLOR_RED);
394
395 break;
396 }
